# Function to log requests, count offenders, and send Telegram alerts
def log_request(ip, path, user_agent, warning=False):

    with open(LOG_FILE, 'a') as log_file:
        line = f'{datetime.datetime.now()} - IP: {ip} - Path: {path} - User-Agent: {user_agent}'

# Route: Main sensor data endpoint
@app.route('/sensor/data', methods=['GET'])
def get_sensor_data():
    ip = request.remote_addr
    path = request.path
    user_agent = request.headers.get('User-Agent')
    log_request(ip, path, user_agent)  # Log normal request

    try:
        temperature = dhtDevice.temperature
        humidity = dhtDevice.humidity

        if humidity is not None and temperature is not None:
            return jsonify({'temperature': temperature, 'humidity': humidity})  # Return sensor data as JSON
        else:
            return jsonify({'error': 'Sensor read failed'}), 500               # Return error if sensor fails

    except RuntimeError as error:
        return jsonify({'error': str(error)}), 500                            # Handle sensor exceptions
